Project: Weather App

Build a script that fetches live weather data for any city.

Project Overview

In this project, you will use the requests library to make a call to the OpenWeatherMap API, parse the JSON data, and display the current temperature and conditions to the user.

Prerequisites: You will need to sign up for a free API key at openweathermap.org.

The Code

Copy this script, replace the dummy API key with your real one, and run it in your terminal.

import requests

def get_weather(city_name):
    api_key = "YOUR_FREE_API_KEY_HERE"
    base_url = "http://api.openweathermap.org/data/2.5/weather?"
    
    # Construct the final API URL
    complete_url = base_url + "q=" + city_name + "&appid=" + api_key + "&units=metric"
    
    # Make the GET request
    response = requests.get(complete_url)
    data = response.json()
    
    # Check if city is found
    if data["cod"] != "404":
        main_data = data["main"]
        weather_desc = data["weather"][0]["description"]
        
        print(f"\nWeather in {city_name}:")
        print(f"Temperature: {main_data['temp']}°C")
        print(f"Humidity: {main_data['humidity']}%")
        print(f"Description: {weather_desc.capitalize()}")
    else:
        print("City Not Found!")

# Get user input
user_city = input("Enter city name: ")
get_weather(user_city)
